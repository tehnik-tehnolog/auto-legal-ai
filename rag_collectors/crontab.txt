# Путь к интерпретатору и скрипту (замените /opt/rag_collectors на ваш путь)
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/sbin:/bin:/usr/sbin:/usr/bin
PROJECT_DIR=/opt/rag_collectors
PYTHON=/opt/rag_collectors/venv/bin/python

# 1. Ассортимент доступных машин (raskatof.ru) — РАЗ В НЕДЕЛЮ (каждый понедельник в 04:00)
0 4 * * 1 cd $PROJECT_DIR && $PYTHON runner.py --collector cars >> logs/cron_cars.log 2>&1

# 2. Автомобильные новости и аналитика — РАЗ В НЕДЕЛЮ (каждый понедельник в 05:00)
0 5 * * 1 cd $PROJECT_DIR && $PYTHON runner.py --collector news >> logs/cron_news.log 2>&1

# 3. Судебная практика ВС РФ (споры по авто) — РАЗ В МЕСЯЦ (1-е число каждого месяца в 03:00)
0 3 1 * * cd $PROJECT_DIR && $PYTHON runner.py --collector vsrf >> logs/cron_vsrf.log 2>&1

# 4. Нормативная база (ГК РФ / НК РФ) — РАЗ В 3 МЕСЯЦА (1-е число января, апреля, июля, октября в 02:00)
0 2 1 1,4,7,10 * cd $PROJECT_DIR && $PYTHON runner.py --collector laws >> logs/cron_laws.log 2>&1
